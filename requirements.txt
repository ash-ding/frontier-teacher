# Direct dependencies. Versions are the ones this project was verified on
# (requirements.lock.txt has the full resolved set).
#
# One environment both evaluates and trains. Install order matters, and two of
# these pins are load-bearing rather than cautious.

# vllm FIRST and alone: it selects the CUDA-matched torch build (2.13.0+cu130
# here). Installing torch separately, or letting a later resolution step move
# it, is the usual way this environment breaks.
vllm==0.27.1

# GRPO and the rest of the RL stack. Pulls transformers back to 5.10.x and
# antlr4 to 4.9.3 via hydra-core - both fine, see the math-verify note below.
verl==0.9.0

# Symbolic answer equivalence for MATH and HMMT, whose answers are exact forms
# rather than integers. The extra must MATCH the antlr4 runtime the environment
# ends up with: verl's hydra-core pins 4.9.3, so use antlr4_9_3, not
# antlr4_13_2. A mismatch does not raise - the LaTeX parser degrades silently
# and depresses every score. Verified with src/calibrate_grading.py.
math-verify[antlr4_9_3]==0.9.0

datasets==5.0.1
accelerate==1.14.0
pandas==3.0.5
tabulate==0.10.0
PyYAML==6.0.3
